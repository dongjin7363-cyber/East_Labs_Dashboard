import sys, subprocess, pkgutil
def _ensure(pkg):
    try:
        import importlib.util
        if importlib.util.find_spec(pkg) is None:
            subprocess.check_call([sys.executable, "-m", "pip", "install", pkg, "--quiet"])
    except Exception:
        # 구버전 호환
        if pkgutil.find_loader(pkg) is None:
            subprocess.check_call([sys.executable, "-m", "pip", "install", pkg, "--quiet"])
_ensure("Pillow")
_ensure("requests")

# --- 2) 유틸 함수 ---
def parse_ticker_from_url(url: str):
    """Finviz chart URL에서 t=티커 추출"""
    try:
        q = parse_qs(urlparse(url).query)
        t = q.get("t", [None])[0]
        return t.upper() if t else None
    except Exception:
        return None

def ensure_domain(url: str, domain: str) -> str:
    """요청 URL의 도메인을 지정 도메인으로 교체"""
    u = urlparse(url)
    return urlunparse((u.scheme or "https", domain, u.path, u.params, u.query, u.fragment))

def download_image(url: str, out_dir=OUT_DIR, sleep=0.8, retries=2):
    """
    Finviz 차트 다운로드 (도메인 폴백 + 헤더 + 간단 재시도)
    반환: (저장경로, 티커)
    """
    ticker = parse_ticker_from_url(url) or "TICKER"
    # 확장자는 finviz가 png/gif 혼재 가능하지만, 저장은 png로 통일
    fname = os.path.join(out_dir, f"{ticker}.png")

    # 이미 받아둔 파일 재사용
    if os.path.exists(fname) and os.path.getsize(fname) > 0:
        return fname, ticker

    headers = {
        "User-Agent": USER_AGENT,
        "Referer": f"https://finviz.com/quote.ashx?t={ticker}"
    }
    sess = requests.Session()
    sess.headers.update({"User-Agent": USER_AGENT})

    last_err = None
    for domain in DOMAINS:
        req_url = ensure_domain(url, domain)
        for attempt in range(retries + 1):
            try:
                r = sess.get(req_url, headers=headers, timeout=15)
                ct = r.headers.get("Content-Type", "")
                if r.status_code == 200 and str(ct).startswith("image"):
                    with open(fname, "wb") as f:
                        f.write(r.content)
                    time.sleep(sleep)
                    return fname, ticker
                else:
                    last_err = f"{domain} HTTP {r.status_code} ct={ct}"
            except Exception as e:
                last_err = f"{domain} {e}"
            time.sleep(0.6)  # 간단 백오프
    raise RuntimeError(f"Download failed for {ticker}: {last_err}")

def make_exact_fit(im: PILImage.Image, w: int, h: int, bg=(255,255,255)):
    """
    비율 유지로 (w,h) 박스 안에 맞추고 여백은 흰색 패딩으로 채운 새 이미지 반환.
    (현재 코드는 화면 표시만 축소하므로, 이 함수는 필요 시에만 사용)
    """
    im = im.convert("RGB")
    im_fit = ImageOps.contain(im, (w, h), PILImage.LANCZOS)
    canvas = PILImage.new("RGB", (w, h), bg)
    canvas.paste(im_fit, ((w - im_fit.width)//2, (h - im_fit.height)//2))
    return canvas

# --- 5) 코랩에 80% 크기로 출력 ---
for path, label in saved:
    # 캡션
    display(HTML(f"<div style='margin:12px 0 6px; font-weight:600;'>{label}</div>"))

    # 원본 사이즈 확인
    with PILImage.open(path) as im:
        w, h = im.size

    # 화면 표시 크기 (원본의 80%)
    disp_w = max(1, int(round(w * SCALE_FACTOR)))
    disp_h = max(1, int(round(h * SCALE_FACTOR)))

    if RESIZE_FILES:
        # 파일 자체를 80%로 리사이즈해서 저장 후 표시 (용량↓)
        small_path = os.path.join(
            THUMB_DIR,
            os.path.splitext(os.path.basename(path))[0] + f"_p{int(SCALE_FACTOR*100)}.jpg"
        )
        with PILImage.open(path) as im:
            im_small = im.convert("RGB").resize((disp_w, disp_h), PILImage.LANCZOS)
            im_small.save(small_path, quality=90)
        display(IPyImage(filename=small_path))
    else:
        # 원본 파일은 그대로 두고, 화면 표시만 80% 축소
        display(IPyImage(filename=path, width=disp_w, height=disp_h))
